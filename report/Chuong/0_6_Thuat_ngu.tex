    \Large{\textbf{DANH MỤC CÁC THUẬT NGỮ}}\\
\end{center}
\vspace{1cm}

\begin{longtable}{l p{11cm}}
	\hline
   \textbf{Thuật ngữ}  & \textbf{Ý nghĩa/Giải thích} \\ \hline 
	\textbf{Browser} & Trình duyệt web sử dụng để hiển thị giao diện cho người dùng cuối.\\
	\textbf{Cache memory} & Bộ nhớ đệm lưu trữ tạm thời dữ liệu có tần suất truy cập cao để tăng tốc độ phản hồi.\\
	\textbf{E-commerce} & Thương mại điện tử – mô hình kinh doanh và mua bán hàng hóa trực tuyến.\\
	\textbf{Multi-vendor} & Đa nhà bán hàng – mô hình cho phép nhiều đối tác cùng đăng ký mở gian hàng và bán sản phẩm.\\
	\textbf{Atomic Operation} & Toán tử nguyên tử – các câu lệnh thực hiện trọn vẹn hoặc thất bại hoàn toàn, không thể bị xen ngang.\\
	\textbf{Overselling} & Hiện tượng bán quá mức tồn kho thực tế do tranh chấp đồng thời.\\
	\textbf{Race Condition} & Tranh chấp đồng thời – tình trạng các tiến trình cạnh tranh tài nguyên dẫn đến sai lệch dữ liệu.\\
	\textbf{Idempotency} & Tính bất biến/tính đồng nhất – tính chất đảm bảo một yêu cầu lặp lại nhiều lần vẫn cho ra cùng một kết quả duy nhất.\\
	\textbf{Time Decay} & Suy hao theo thời gian – thuật toán giảm dần trọng số điểm quan tâm đối với các tương tác cũ.\\
	\textbf{Collaborative Filtering} & Lọc cộng tác – thuật toán khuyến nghị dựa trên hành vi tương đồng của nhóm người dùng.\\
	\textbf{Content-based Filtering} & Lọc theo nội dung – thuật toán gợi ý dựa trên đặc trưng phân loại của sản phẩm khách hàng yêu thích.\\
	\textbf{Hybrid Recommendation} & Khuyến nghị lai – sự kết hợp đa phương pháp lọc để tăng độ chính xác gợi ý và sửa lỗi Khởi đầu lạnh.\\
	\textbf{Rate limiting} & Giới hạn tần suất truy cập API để ngăn chặn tấn công từ chối dịch vụ hoặc brute-force.\\
	\textbf{Token Rotation} & Xoay vòng khóa phiên – cơ chế thu hồi và cấp mới mã xác thực tự động để bảo vệ phiên làm việc.\\
	\textbf{Deny-list} & Danh sách đen/danh sách từ chối – tập hợp các mã token bị thu hồi hiệu lực trước thời hạn.\\
	\textbf{Object Storage} & Lưu trữ đối tượng – dịch vụ lưu trữ dữ liệu phi cấu trúc (như hình ảnh) hiệu năng cao.\\
	\textbf{Webhook} & Cơ chế phản hồi gọi ngược HTTP tự động từ dịch vụ bên thứ ba (như Stripe) về máy chủ khi có sự kiện thanh toán.\\
	\textbf{Server-Sent Events} & Luồng đẩy sự kiện từ máy chủ – kênh truyền dữ liệu một chiều thời gian thực tiêu chuẩn HTML5.\\
	\textbf{Real-time} & Xử lý thời gian thực – phản hồi và truyền dẫn thông tin tức thời qua kết nối Socket.\\
    \hline
\end{longtable}
